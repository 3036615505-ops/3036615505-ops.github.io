/* veil：全屏纸色兜底遮罩，负责接住跳转那一帧（opacity，合成层，不重绘）。
   只是一块纯色，不是「一页纸」。 */
.page-transition-veil {
  position: absolute;
  inset: 0;
  z-index: 1;
  background: #f0e8d6;
  opacity: 0;
  pointer-events: none;
}

@keyframes veilIn {
  from { opacity: 0; }
  to   { opacity: 1; }
}

@keyframes veilOut {
  from { opacity: 1; }
  to   { opacity: 0; }
}

/* veil 只做兜底：跳转那一帧由它接住，防止闪旧页面。
   延后进，别盖住书本翻开的过程。 */
.page-transition.flip-out .page-transition-veil {
  animation: veilIn 80ms ease-out 340ms forwards;
}

.page-transition.flip-in .page-transition-veil {
  animation: veilOut 160ms ease-in forwards;
}

/* 翻书/放大时底部小字跟随淡出，别在正文露出后还挂着 */
.page-transition.flip-out .transition-text,
.page-transition.flip-in .transition-text {
  animation: bookTextFade 170ms ease forwards;
}

@keyframes bookTextFade {
  0%, 50% { opacity: 1; }
  100% { opacity: 0; }
}

/* 翻书期间容器不再做 opacity 淡入淡出，否则「先出雾、再出书」多一拍延迟感 */
.page-transition.flip-out,
.page-transition.flip-in {
  transition: none;
  opacity: 1;
  /* 容器自身透明：遮挡由 veil 负责。容器若是实底，会和 veil 一起形成
     两层色块，翻书时像蒙了层脏纱 */
  background: transparent;
}

/* 「减弱动态效果」：不做翻书放大，保留直来直去 */

  .page-transition.flip-in .book-page,
  .page-transition.flip-out .transition-text,
  .page-transition.flip-in .transition-text {
    animation: none;
    transition: none;
  }
  /* 没有动画时遮挡仍然要有，否则跳转那一帧会闪旧页面 */
  .page-transition.flip-out .page-transition-veil {
    opacity: 1;
    animation: none;
  }
  /* 封面仍要盖住跳转（铺满的是封面图），否则旧页面会闪出来 */
  .page-transition.flip-out .book-cover {
    opacity: 1;
  }
  .page-transition.flip-in .book-cover {
    opacity: 0;
  }
}
